GREEN = '\033[92m'
RED   = '\033[91m'
BOLD  = '\033[1m'
DIM   = '\033[2m'
RESET = '\033[0m'

def fmt_ans(ans: bool) -> str:
    word = 'yes' if ans else 'no'
    color = GREEN if ans else RED
    return f'{color}{word}{RESET}'

def print_qa_list(qa_list, indent=2):
    pad = ' ' * indent
    for qa in qa_list:
        print(f'{pad}Q: {qa["question"]} → A: {fmt_ans(qa["answer"])}')

def is_reflexive(question: str) -> bool:
    """Return True if the subject and object block in the question are identical."""
    q = question.lower().rstrip('?')
    for sep in (' close to ', ' touching '):
        if sep in q:
            before, after = q.split(sep, 1)
            subj = before.removeprefix('is the ').strip()
            obj  = after.removeprefix('the ').strip()
            return subj == obj
    return False
